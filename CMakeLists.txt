cmake_minimum_required(VERSION 3.24)
project(NimbyTco VERSION 0.5.2 LANGUAGES CXX)
find_package(Qt6 6.5 REQUIRED COMPONENTS Quick QuickControls2 Network)
find_package(NimbyRailsFranceSDK 0.7.1 CONFIG REQUIRED)
qt_standard_project_setup(REQUIRES 6.5)
qt_add_executable(NimbyTco main.cpp backend.cpp backend.h mapitem.cpp mapitem.h updater.cpp updater.h sdkclient.cpp sdkclient.h)
qt_add_qml_module(NimbyTco URI NimbyTco QML_FILES Main.qml Panel.qml TrainFilter.js)
target_compile_features(NimbyTco PRIVATE cxx_std_20)
target_compile_definitions(NimbyTco PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_definitions(NimbyTco PRIVATE TCO_VERSION="${PROJECT_VERSION}" NIMBY_SDK_BUILD)
get_target_property(SDK_PUBLIC_HEADERS NimbyRailsFranceSDK::SDK INTERFACE_INCLUDE_DIRECTORIES)
target_include_directories(NimbyTco PRIVATE ${SDK_PUBLIC_HEADERS})
target_link_libraries(NimbyTco PRIVATE Qt6::Quick Qt6::QuickControls2 Qt6::Network)
set_target_properties(NimbyTco PROPERTIES WIN32_EXECUTABLE TRUE)
option(NRF_DEPLOY_QT "Deploy the Qt runtime for local Run/Debug" ON)
if(WIN32 AND NRF_DEPLOY_QT)
    find_program(NRF_WINDEPLOYQT windeployqt HINTS "${Qt6_DIR}/../../../bin" REQUIRED)
    add_custom_command(TARGET NimbyTco POST_BUILD
        COMMAND "${NRF_WINDEPLOYQT}" --release --no-translations --qmldir "${CMAKE_CURRENT_SOURCE_DIR}" "$<TARGET_FILE:NimbyTco>"
        COMMAND ${CMAKE_COMMAND} -E copy_if_different
            "$<TARGET_FILE_DIR:NimbyRailsFranceSDK::SDK>/libwinpthread-1.dll" "$<TARGET_FILE_DIR:NimbyTco>"
        VERBATIM)
endif()
add_custom_command(TARGET NimbyTco POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
    "$<TARGET_FILE:NimbyRailsFranceSDK::SDK>" "$<TARGET_FILE_DIR:NimbyTco>")

include(CTest)
if(BUILD_TESTING)
    qt_add_executable(tco_signal_image_tests tests/signal_image.cpp)
    target_link_libraries(tco_signal_image_tests PRIVATE Qt6::Gui)
    add_test(NAME signal_image COMMAND tco_signal_image_tests)
    qt_add_executable(tco_train_filter_tests tests/train_filter.cpp)
    target_link_libraries(tco_train_filter_tests PRIVATE Qt6::Qml)
    target_compile_definitions(tco_train_filter_tests PRIVATE TCO_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
    add_test(NAME train_filter COMMAND tco_train_filter_tests)
    qt_add_executable(tco_update_tests tests/update_manifest.cpp updater.cpp updater.h)
    target_link_libraries(tco_update_tests PRIVATE Qt6::Network)
    target_compile_features(tco_update_tests PRIVATE cxx_std_20)
    add_test(NAME update_manifest COMMAND tco_update_tests)
    qt_add_executable(tco_path_refresh_tests tests/path_refresh.cpp backend.cpp backend.h mapitem.cpp mapitem.h)
    target_compile_features(tco_path_refresh_tests PRIVATE cxx_std_20)
    target_compile_definitions(tco_path_refresh_tests PRIVATE NIMBY_SDK_BUILD WIN32_LEAN_AND_MEAN NOMINMAX)
    get_target_property(SDK_PUBLIC_HEADERS NimbyRailsFranceSDK::SDK INTERFACE_INCLUDE_DIRECTORIES)
    target_include_directories(tco_path_refresh_tests PRIVATE ${SDK_PUBLIC_HEADERS})
    target_link_libraries(tco_path_refresh_tests PRIVATE Qt6::Quick)
    add_test(NAME path_refresh COMMAND tco_path_refresh_tests)
    set_tests_properties(path_refresh PROPERTIES TIMEOUT 30)
endif()
